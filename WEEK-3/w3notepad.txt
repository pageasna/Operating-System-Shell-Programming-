#include <stdio.h>
int main() {
	int n, i, j;
	
	printf("Enter number of processes: ");
	if(scanf("%d",&n)!=1 || n < 0) {
		printf("Invalid number of processes.");
		return 1;
	}
	float a[n], temp;
	printf("Enter arrival time for the processes:\n");
	for(i=0; i<n; i++) {
		printf("P%d: ",i);
		if(scanf("%f", &a[i])!=1 || a[i] < 0) {
			printf("Invalid arrival time!");
			return 1;
		}
	}
	for(i=0; i< n-1; i++) {
		for(int j=0; j<n-i-1; j++){
			if(a[j]>a[j+1]) {
				temp=a[j];
				a[j]=a[j+1];
				a[j+1]=temp;
			}
		}
	}
	printf("Processes in ascending order of arrival time:\n");
	for(i=0; i<n; i++) {
		printf("P%d : %.2f -> ",i,a[i]);
	}
}

#include <stdio.h>
int main() {
	int n;
	float temp, sum=0, average;
	printf("Enter number of processes: ");
	if(scanf("%d",&n)!=1 || n < 0) {
		printf("Invalid number of processes.");
		return 1;
	}

	printf("Enter Burst time for the processes:\n");
	for(int i=1; i<=n; i++) {
		if(scanf("%f",&temp)!=1 || temp < 0) {
			printf("Invalid Burst time!");
			return 1;
		}
		sum+=temp;
	}

	average = sum/n;
	printf("Sum of burst time: %.2f\n",sum);
	printf("Average of burst time: %.2f\n",average);
	return 0;
}

#include <stdio.h>
struct process {
	int id;
	float arrival;
};

int main() {
	int n, i, j;
	struct process p[100], temp;
	printf("Enter number of processes: ");
	if(scanf("%d",&n)!=1 || n < 0) {
		printf("Invalid number of processes.");
		return 1;
	}
	
	printf("Enter arrival time for the processes:\n");
	for(i=0; i<n; i++) {
		p[i].id = i+1;
		printf("P%d: ",p[i].id);
		if(scanf("%f", &p[i].arrival)!=1 || p[i].arrival < 0) {
			printf("Invalid arrival time!");
			return 1;
		}
	}
	for(i=0; i< n-1; i++) {
		for(int j=0; j<n-i-1; j++){
			if(p[j].arrival>p[j+1].arrival) {
				temp=p[j];
				p[j]=p[j+1];
				p[j+1]=temp;
			}
		}
	}
	printf("Processes in ascending order of arrival time:\n");
	for(i=0; i<n; i++) {
		printf("P%d : %.2f \n",p[i].id,p[i].arrival);
	}
	return 0;
}
