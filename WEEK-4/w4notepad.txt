#include <stdio.h>

// 1. Define Process Structure
struct Process {
    int pid;
    double arrival_time;
    double burst_time;
};

int main() {
    struct Process p1;
    printf("Enter Process ID :");
    if(scanf("%d",&p1.pid)!=1||p1.pid<=0){
    	printf("Invalid Input!!");
        return 1;
    }
    printf("Enter Arrival Time :");
    if(scanf("%lf",&p1.arrival_time)!=1||p1.arrival_time<=0){
    	printf("Invalid Input!!");
        return 1;
    }    
    printf("Enter Burst_time :");
    if(scanf("%lf",&p1.burst_time)!=1||p1.burst_time<=0){
    	printf("Invalid Input!!");
        return 1;
    }
    printf("Process ID: %d\n", p1.pid);
    printf("Arrival Time: %.2lf\n", p1.arrival_time);
    printf("Burst Time: %.2lf\n", p1.burst_time);
    
    return 0;
}

#include <stdio.h>

#define MAX 100

// 1. Define Process Structure
struct Process {
    int pid;
    double arrival_time;
    double burst_time;
    double completion_time;
};

// Function to calculate completion time based on task 3 formula
void calculateCompletionTime(struct Process proc[], int n) {
    for (int i = 0; i < n; i++) {
        proc[i].completion_time = proc[i].arrival_time + proc[i].burst_time;
    }
}

// Function to take input from user
void inputProcesses(struct Process proc[], int n) {
    for (int i = 0; i < n; i++) {
        proc[i].pid = i + 1;
        printf("\n--- Process %d ---\n", proc[i].pid);
        
        printf("Enter Arrival Time: ");
        scanf("%lf", &proc[i].arrival_time);
        
        printf("Enter Burst Time: ");
        scanf("%lf", &proc[i].burst_time);
    }
}

// Function to display processes with completion time
void displayProcesses(struct Process proc[], int n) {
    printf("\n----------------------------------------------------\n");
    printf("PID\tArrival Time\tBurst Time\tCompletion Time\n");
    printf("----------------------------------------------------\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%.2lf\t\t%.2lf\t\t%.2lf\n", 
               proc[i].pid, 
               proc[i].arrival_time, 
               proc[i].burst_time, 
               proc[i].completion_time);
    }
    printf("----------------------------------------------------\n");
}

int main() {
    int n;
    struct Process proc[MAX];

    printf("Enter number of processes: ");
    scanf("%d", &n);
    inputProcesses(proc, n);
    calculateCompletionTime(proc, n);
    displayProcesses(proc, n);

    return 0;
}

#include <stdio.h>


struct Process{
    int pid;
    double arrival_time;
    double burst_time;
};

int main(void)
{
    int n;

    printf("Enter number of processes: ");
    scanf("%d",&n);

    struct Process processes[n];

    for (int i = 0; i < n; i++)
    {
        printf("\nEnter details for Process %d\n", i + 1);

        printf("PID: ");
        scanf("%d",&processes[i].pid);

        printf("Arrival Time: ");
         scanf("%lf",&processes[i].arrival_time);

        printf("Burst Time: ");
         scanf("%lf",&processes[i].burst_time);
    }
   
    int shortest = 0;

    for (int i = 1; i < n; i++)
    {
        if (processes[i].burst_time < processes[shortest].burst_time)
        {
            shortest = i;
        }
    }

    printf("\nShortest Job Next:\n");

    printf("PID: %d\n", processes[shortest].pid);
    printf("Arrival Time: %.2lf\n", processes[shortest].arrival_time);
    printf("Burst Time: %.2lf\n", processes[shortest].burst_time);

    return 0;
}
