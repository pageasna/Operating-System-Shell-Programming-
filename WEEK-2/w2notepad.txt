#include <stdio.h>

int main()
{
    float busy_time, idle_time, utilization;

    printf("Enter busy time: ");
    if(scanf("%f", &busy_time)!=1 || busy_time<0) {
	printf("Invalid busy time!\n");
	return 1;
    }

    printf("Enter idle time: ");
    if(scanf("%f", &idle_time)!=1 || idle_time<0) {
	printf("Invalid idle time!\n");
	return 1;
    }
    double total_time = busy_time + idle_time;
    if(total_time==0) {
	printf("CPU Utilization is undefined because total time is zero.\n");
	return 1;
    }
    utilization = (busy_time / total_time) * 100;
    printf("CPU Utilization = %.2f%%\n", utilization);

    return 0;
}

#include <stdio.h>

int main()
{
    int priority[5];
    int i, highest = 0;

    for (i = 0; i < 5; i++)
    {
        printf("Enter priority of Process %d (1=High, 2=Medium, 3=Low): ",
               i + 1);
        scanf("%d", &priority[i]);
    }

    if(priority[i]!=1 || priority[i]!=2 || priority[i]!=3) {
	printf("Invalid priority! Please enter 1=High, 2=Medium or 3=Low.\n");
	return 1;
    }

    for (i = 1; i < 5; i++)
    {
        if (priority[i] < priority[highest])
            highest = i;
    }

    printf("\nProcess %d will run first.\n", highest + 1);
    printf("Priority = %d\n", priority[highest]);

    return 0;
}

#include <stdio.h>

int main()
{
    int processes;
    float total_time, throughput;

    printf("Enter number of processes: ");
    if(scanf("%d", &processes)!=1 || processes<0) {
	printf("Invalid number of processes.\n");
	return 1;
    }

    printf("Enter total time: ");
    if(scanf("%f", &total_time)!=1 || total_time<=0) {
	printf("Invalid total time.\n");
	return 1;
    }

    throughput = processes / total_time;

    printf("Throughput = %.2f processes/unit time\n", throughput);

    return 0;
}

#include <stdio.h>

int main()
{
    int n, i;
    char process[50][20];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        printf("Enter process %d name: ", i + 1);
        scanf("%19s", process[i]);
    }

    printf("\nReady Queue / Execution Order:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s", process[i]);

        if (i < n - 1)
            printf(" -> ");
    }

    printf("\n");

    return 0;
}
